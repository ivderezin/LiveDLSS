cmake_minimum_required(VERSION 3.25)
project(LiveDLSS.NeuralRendering VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The NR bridge requires Windows x64.")
endif()
set(NGX_SDK_DIR "$ENV{NGX_SDK_DIR}" CACHE PATH "NVIDIA DLSS SDK root (include/nvsdk_ngx_params.h)")
find_path(NGX_INCLUDE_DIR nvsdk_ngx_params.h PATHS "${NGX_SDK_DIR}/include" "${NGX_SDK_DIR}" NO_DEFAULT_PATH REQUIRED)
add_library(LiveDLSS.NeuralRendering SHARED src/ldls_nr.cpp)
set(BRIDGE_FILENAME "nvngx.dll_LiveDLSS.NeuralRendering.dll")
configure_file(version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
target_sources(LiveDLSS.NeuralRendering PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
target_compile_definitions(LiveDLSS.NeuralRendering PRIVATE LIVEDLSS_VERSION="${PROJECT_VERSION}")
target_compile_features(LiveDLSS.NeuralRendering PRIVATE cxx_std_20)
target_include_directories(LiveDLSS.NeuralRendering PRIVATE include "${NGX_INCLUDE_DIR}")
target_compile_definitions(LiveDLSS.NeuralRendering PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(LiveDLSS.NeuralRendering PRIVATE d3d12 advapi32)
if(MSVC)
  target_compile_options(LiveDLSS.NeuralRendering PRIVATE /W4 /WX /wd4191 /permissive- /utf-8)
endif()
# The NR runtime requires nvngx.dll in the calling module's filename.
set_target_properties(LiveDLSS.NeuralRendering PROPERTIES OUTPUT_NAME "nvngx.dll_LiveDLSS.NeuralRendering" PREFIX "")
set(LIVEDLSS_RUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../runtimes/neural-rendering" CACHE PATH "Bridge install directory")
install(TARGETS LiveDLSS.NeuralRendering RUNTIME DESTINATION "${LIVEDLSS_RUNTIME_DIR}")
